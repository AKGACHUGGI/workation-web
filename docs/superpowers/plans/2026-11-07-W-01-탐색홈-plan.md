`src/components/ui/Dropdown.tsx`:

```tsx
import { useEffect, useId, useRef, useState } from 'react'
import type { KeyboardEvent } from 'react'
import { ChevronDown } from 'lucide-react'

export type DropdownOption<T extends string> = { value: T; label: string }

type Props<T extends string> = {
  options: DropdownOption<T>[]
  value: T
  onChange: (value: T) => void
  ariaLabel: string
  className?: string
}

/** 단일 선택 드롭다운. 버튼(aria-haspopup=listbox) + listbox/option. */
export default function Dropdown<T extends string>({
  options,
  value,
  onChange,
  ariaLabel,
  className = '',
}: Props<T>) {
  const [open, setOpen] = useState(false)
  const [active, setActive] = useState(() =>
    Math.max(
      0,
      options.findIndex((o) => o.value === value),
    ),
  )
  const rootRef = useRef<HTMLDivElement>(null)
  const listId = useId()
  const current = options.find((o) => o.value === value) ?? options[0]

  useEffect(() => {
    if (!open) return
    const onDown = (e: MouseEvent) => {
      if (!rootRef.current?.contains(e.target as Node)) setOpen(false)
    }
    document.addEventListener('mousedown', onDown)
    return () => document.removeEventListener('mousedown', onDown)
  }, [open])

  const select = (i: number) => {
    onChange(options[i].value)
    setOpen(false)
  }

  const onKeyDown = (e: KeyboardEvent<HTMLDivElement>) => {
    if (e.key === 'Escape') {
      setOpen(false)
      return
    }
    if (!open) {
      if (e.key === 'Enter' || e.key === ' ' || e.key === 'ArrowDown') {
        e.preventDefault()
        setActive(
          Math.max(
            0,
            options.findIndex((o) => o.value === value),
          ),
        )
        setOpen(true)
      }
      return
    }
    if (e.key === 'ArrowDown') {
      e.preventDefault()
      setActive((i) => Math.min(options.length - 1, i + 1))
    } else if (e.key === 'ArrowUp') {
      e.preventDefault()
      setActive((i) => Math.max(0, i - 1))
    } else if (e.key === 'Enter' || e.key === ' ') {
      e.preventDefault()
      select(active)
    }
  }

  return (
    <div ref={rootRef} className={`relative inline-block ${className}`} onKeyDown={onKeyDown}>
      <button
        type="button"
        aria-haspopup="listbox"
        aria-expanded={open}
        aria-controls={open ? listId : undefined}
        aria-label={`${ariaLabel}: ${current.label}`}
        onClick={() => setOpen((v) => !v)}
        className="border-border-default bg-bg-surface text-body-sm text-text-primary hover:border-border-strong inline-flex h-9 items-center gap-2 rounded-lg border px-3"
      >
        {current.label}
        <ChevronDown size={16} aria-hidden />
      </button>
      {open && (
        <ul
          id={listId}
          role="listbox"
          aria-label={ariaLabel}
          className="border-border-default bg-bg-surface shadow-float absolute top-full right-0 z-20 mt-1 min-w-full rounded-lg border py-1"
        >
          {options.map((opt, i) => {
            const selected = opt.value === value
            return (
              <li
                key={opt.value}
                role="option"
                aria-selected={selected}
                onMouseEnter={() => setActive(i)}
                onClick={() => select(i)}
                className={`text-body-sm cursor-pointer px-3 py-2 whitespace-nowrap ${
                  i === active ? 'bg-bg-subtle' : ''
                } ${selected ? 'text-text-primary font-medium' : 'text-text-secondary'}`}
              >
                {opt.label}
              </li>
            )
          })}
        </ul>
      )}
    </div>
  )
}
```

```ts
import { POPULAR_SORTS, TERRAINS } from '@/features/popular/types'
import type { PopularSort, Terrain } from '@/features/popular/types'

export const DEFAULT_SORT: PopularSort = 'favorites'

export function parseSort(raw: string | null): PopularSort | null {
  if (raw === null) return null
  return (POPULAR_SORTS as readonly string[]).includes(raw) ? (raw as PopularSort) : null
}

function isTerrain(v: string): v is Terrain {
  return (TERRAINS as readonly string[]).includes(v)
}

export function normalizeTerrain(list: readonly Terrain[]): Terrain[] {
  return TERRAINS.filter((t) => list.includes(t))
}

/** 'city,coast,coast' → ['coast','city']. '' → []. 모르는 값 → null */
export function parseTerrain(raw: string | null): Terrain[] | null {
  if (raw === null) return null
  if (raw === '') return []
  const parts = raw.split(',').filter(Boolean)
  if (!parts.every(isTerrain)) return null
  return normalizeTerrain(parts)
}

export function serializeTerrain(list: readonly Terrain[]): string {
  return normalizeTerrain(list).join(',')
}
```

`src/features/popular/useTerrainParam.ts`:

```ts
import { useCallback, useEffect } from 'react'
import { useSearchParams } from 'react-router-dom'
import { parseTerrain, serializeTerrain } from '@/features/popular/params'
import type { Terrain } from '@/features/popular/types'

/**
 * terrain 쿼리 (복수). 빈 배열이면 키를 지운다 (기본값 "없음").
 * 무효 값은 replace 로 키 삭제.
 */
export function useTerrainParam(): [Terrain[], (next: Terrain[]) => void] {
  const [searchParams, setSearchParams] = useSearchParams()
  const raw = searchParams.get('terrain')
  const parsed = parseTerrain(raw)
  const invalid = raw !== null && parsed === null
  const value = parsed ?? []

  useEffect(() => {
    if (!invalid) return
    setSearchParams(
      (prev) => {
        const next = new URLSearchParams(prev)
        next.delete('terrain')
        return next
      },
      { replace: true },
    )
  }, [invalid, setSearchParams])

  const set = useCallback(
    (next: Terrain[]) => {
      setSearchParams((prev) => {
        const sp = new URLSearchParams(prev)
        if (next.length === 0) sp.delete('terrain')
        else sp.set('terrain', serializeTerrain(next))
        return sp
      })
    },
    [setSearchParams],
  )

  return [value, set]
}
```

```ts
import { composite } from '@/features/metric/score'
import type { Weights } from '@/features/metric/types'
import { SORT_BADGE_PREFIX } from '@/features/popular/copy'
import type { PopularRegion, PopularSort, Terrain } from '@/features/popular/types'

/** 정렬 기준별 비교. 새 배열을 돌려준다. */
export function sortPopular(
  items: PopularRegion[],
  sort: PopularSort,
  weights: Weights,
): PopularRegion[] {
  const score = new Map(items.map((i) => [i.id, composite(i.scores, weights)]))
  const byScore = (a: PopularRegion, b: PopularRegion) =>
    (score.get(b.id) ?? 0) - (score.get(a.id) ?? 0)
  const byFav = (a: PopularRegion, b: PopularRegion) => b.favoriteCount - a.favoriteCount
  const cmp: Record<PopularSort, (a: PopularRegion, b: PopularRegion) => number> = {
    favorites: (a, b) => byFav(a, b) || byScore(a, b),
    views: (a, b) => b.viewCount - a.viewCount || byFav(a, b),
    score: (a, b) => byScore(a, b) || byFav(a, b),
  }
  return [...items].sort(cmp[sort])
}

/** 지형 필터. 빈 필터는 모두 통과, 아니면 하나라도 겹치면 통과 */
export function terrainMatch(item: { terrain: Terrain[] }, terrains: Terrain[]): boolean {
  if (terrains.length === 0) return true
  return item.terrain.some((t) => terrains.includes(t))
}

/** 정렬 기준 상위 3개 배지 */
export function rankBadge(sort: PopularSort, index: number): string | undefined {
  if (index > 2) return undefined
  return `${SORT_BADGE_PREFIX[sort]} ${index + 1}위`
}
```

`src/features/search/applyIntent.ts`:

```ts
import { serializeTerrain } from '@/features/popular/params'
import type { IntentAxis, SearchIntent } from '@/features/search/types'

/** URL 에 반영하는 축 → 쿼리 키 */
const AXIS_KEY: Partial<Record<IntentAxis, string>> = {
  month: 'month',
  preset: 'preset',
  terrain: 'terrain',
}

/**
 * 파싱 결과를 URL 쿼리에 반영한다. month·preset·terrain 만 (stay·peer 는 태그 표시 전용).
 * override=false(기본): 이미 있는 키는 유지 (진입 시 q 파싱). override=true: 덮어쓴다 (검색 실행).
 */
export function applyIntentToParams(
  intent: SearchIntent,
  current: URLSearchParams,
  opts: { override?: boolean } = {},
): URLSearchParams {
  const next = new URLSearchParams(current)
  const put = (key: string, value: string) => {
    if (!opts.override && next.has(key)) return
    next.set(key, value)
  }
  if (intent.month) put('month', intent.month)
  if (intent.preset) put('preset', intent.preset)
  if (intent.terrain && intent.terrain.length > 0) put('terrain', serializeTerrain(intent.terrain))
  if (next.get('preset') !== 'custom') next.delete('metrics')
  return next
}

/** 조건 태그 × 클릭: 해당 축의 쿼리 키 삭제 */
export function removeIntentAxis(axis: IntentAxis, current: URLSearchParams): URLSearchParams {
  const next = new URLSearchParams(current)
  const key = AXIS_KEY[axis]
  if (key) next.delete(key)
  if (key === 'preset') next.delete('metrics')
  return next
}
```

```ts
import { useMutation } from '@tanstack/react-query'

      } finally {
        clearTimeout(timer)
      }
    },
    onError: () => toast.show(SEARCH_FAIL_MESSAGE),
  })
}
```

```tsx
import { useEffect, useRef, useState } from 'react'
import { useSearchParams } from 'react-router-dom'
import ExampleChips from '@/features/search/components/ExampleChips'
import IntentTags from '@/features/search/components/IntentTags'
import SearchBar from '@/features/search/components/SearchBar'
import { applyIntentToParams, removeIntentAxis } from '@/features/search/applyIntent'
import type { IntentAxis, SearchIntent } from '@/features/search/types'
import { useSearchParse } from '@/features/search/useSearchParse'

const NO_RESULT = '조건을 찾지 못했어요. 예시처럼 시기 · 분위기 · 동네를 적어보세요'
const CLEAR_KEYS = ['q', 'month', 'preset', 'metrics', 'terrain'] as const

/** 검색창 · 예시 칩 · AI가 이해한 조건 태그. URL q 가 있으면 진입 시 1회 파싱(URL 값 우선). */
export default function SearchSection() {
  const [searchParams, setSearchParams] = useSearchParams()
  const q = searchParams.get('q') ?? ''
  const parse = useSearchParse()
  const [intent, setIntent] = useState<SearchIntent | null>(null)
  const [noResult, setNoResult] = useState(false)
  const parsedOnEntry = useRef(false)

  const run = (text: string, override: boolean) => {
    parse.mutate(text, {
      onSuccess: (result) => {
        setIntent(result.tags.length ? result : null)
        setNoResult(result.tags.length === 0)
        setSearchParams((prev) => {
          const next = applyIntentToParams(result, prev, { override })
          next.set('q', text)
          return next
        })
      },
    })
  }

  useEffect(() => {
    if (parsedOnEntry.current || !q) return
    parsedOnEntry.current = true
    run(q, false)
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [])

  const removeTag = (axis: IntentAxis, label: string) => {
    setSearchParams((prev) => removeIntentAxis(axis, prev))
    setIntent((cur) => {
      if (!cur) return cur
      const tags = cur.tags.filter((t) => !(t.axis === axis && t.label === label))
      return tags.length ? { ...cur, tags } : null
    })
  }

  const clear = () => {
    setIntent(null)
    setNoResult(false)
    setSearchParams((prev) => {
      const next = new URLSearchParams(prev)
      for (const k of CLEAR_KEYS) next.delete(k)
      return next
    })
  }

  return (
    <section className="flex flex-col gap-4">
      <div>
        <h1 className="text-heading-2xl text-text-primary">어떤 워케이션을 찾고 있나요?</h1>
        <p className="text-body-sm text-text-secondary mt-1">
          LG U+ 유동·서비스인구 데이터로 12개 시군구의 시기·동네 성격을 비교합니다.
        </p>
      </div>
      <SearchBar key={q} initial={q} pending={parse.isPending} onSubmit={(t) => run(t, true)} />
      <ExampleChips onPick={(t) => run(t, true)} />
      {intent && <IntentTags tags={intent.tags} onRemove={removeTag} onClear={clear} />}
      {noResult && <p className="text-body-sm text-text-secondary">{NO_RESULT}</p>}
    </section>
  )
}
```

```tsx
export default function CategoryStrip() {
  const [terrain, setTerrain] = useTerrainParam()
  const [preset, setPreset] = usePresetParam()

  const isActive = (a: CategoryAction) =>
    (a.kind === 'terrain' && terrain.includes(a.value)) ||
    (a.kind === 'preset' && preset === a.value)

  const act = (a: CategoryAction) => {
    if (a.kind === 'terrain') {
      setTerrain(
        terrain.includes(a.value) ? terrain.filter((t) => t !== a.value) : [...terrain, a.value],
      )
    } else if (a.kind === 'preset') {
      setPreset(a.value)
    }
  }

  return (
    <nav aria-label="카테고리" className="border-border-default flex items-center gap-8 border-b">
      {CATEGORIES.map(({ key, label, icon: Icon, action }) => {
        const active = isActive(action)
        const decorative = action.kind === 'none'
        return (
          <button
            key={key}
            type="button"
            aria-pressed={decorative ? undefined : active}
            aria-disabled={decorative || undefined}
            onClick={decorative ? undefined : () => act(action)}
            className={`text-caption-segment -mb-px flex flex-col items-center gap-1 border-b-2 pb-3 ${
              active
                ? 'border-text-primary text-text-primary'
                : 'text-text-secondary border-transparent'
            } ${decorative ? 'cursor-default' : 'hover:text-text-primary'}`}
          >
            <Icon size={22} aria-hidden />
            {label}
          </button>
        )
      })}
    </nav>
  )
}
```

```ts
import { useCallback, useMemo } from 'react'
import { useSearchParams } from 'react-router-dom'
import { PRESET_WEIGHTS } from '@/features/metric/constants'
import { parseMetrics, serializeMetrics } from '@/features/metric/params'
import type { MetricKey, PresetKey, Weights } from '@/features/metric/types'
import { DEFAULT_CUSTOM_METRICS } from '@/features/region/presetMode'
import { customWeights } from '@/features/tier/buildTiers'
import { usePresetParam } from '@/hooks/useQueryState'

/**
 * preset·metrics 쿼리 → 가중치. W-02 의 useWeights 와 같은 시그니처.
 * 기본 선택은 W-03 usePresetMode 와 같은 DEFAULT_CUSTOM_METRICS.
 * selected·weights 는 URL 값이 같으면 참조를 유지한다 (히트맵 폴리곤 getStyle 재생성 방지).
 */
export function useWeights() {
  const [preset, setPreset] = usePresetParam()
  const [searchParams, setSearchParams] = useSearchParams()
  const rawMetrics = searchParams.get('metrics')
  const selected = useMemo(() => parseMetrics(rawMetrics) ?? DEFAULT_CUSTOM_METRICS, [rawMetrics])
  const setSelected = useCallback(
    (keys: MetricKey[]) => {
      setSearchParams((prev) => {
        const next = new URLSearchParams(prev)
        next.set('preset', 'custom')
        next.set('metrics', serializeMetrics(keys))
        return next
      })
    },
    [setSearchParams],
  )
  const weights = useMemo<Weights>(
    () => (preset === 'custom' ? customWeights(selected) : PRESET_WEIGHTS[preset]),
    [preset, selected],
  )
  return { preset, setPreset: setPreset as (p: PresetKey) => void, selected, setSelected, weights }
}
```

```tsx
  items: PopularRegion[]
  month: MonthParam
  hoverId: string | null
  onHover: (id: string | null) => void
}

/** 인기 지역 카드 3열. 정렬·필터·점수·등급은 여기서 계산 */

    const grades = toGrades(sorted.map((i) => composite(i.scores, weights)))
    const withGrade = sorted.map((item, i) => ({ item, grade: grades[i] }))
    return withGrade.filter(({ item }) => terrainMatch(item, terrain))
  }, [items, sort, weights, terrain])

  return (
    <section className="flex flex-col gap-6">
      <PopularHeader
        count={items.length}
        sort={sort}
        onSortChange={setSort}
        preset={preset}
        onPresetChange={setPreset}
        selected={selected}
        onSelectedChange={setSelected}
      />
      {list.length === 0 ? (
        <PopularEmpty onClearFilter={() => setTerrain([])} />
      ) : (
        <div data-testid="popular-grid" className="grid grid-cols-3 gap-6">

              photoUrl={item.photoUrl}

              meta1={item.tags.join(' · ')}
              meta2={`즐겨찾기 ${item.favoriteCount.toLocaleString('ko-KR')} · 조회 ${formatCount(item.viewCount)} · ${monthLabel(month)}`}

                grade: metricGrade(item.scores[m]),

              onToggleFavorite={() => toggle('regions', item.id)}
              href={carryQuery(`/regions/${item.id}`, current)}
              highlighted={hoverId === item.id}
              onMouseEnter={() => onHover(item.id)}
              onMouseLeave={() => onHover(null)}
            />
          ))}
        </div>
      )}
      <p className="text-caption-xs text-text-tertiary">{POPULAR_NOTE}</p>

  )
}
```

```ts
import type { TierScope } from '@/features/tier/types'
import { useTierQuery } from '@/features/tier/useTierQuery'
import { carryQuery } from '@/hooks/carryQuery'

export type HeatEntry = {
  id: string
  name: string
  score: number
  grade: Grade
  step: HeatStep
  dimmed: boolean
  href: string
}

type Args = {
  month: MonthParam
  scope: TierScope
  sgg: string | null
  metric: HeatMetric
  weights: Weights
  terrain: Terrain[]
}

/** tier 응답 → 히트맵 항목. 행정동 범위는 선택한 시군구 안의 동만. */
export function useHeatEntries({ month, scope, sgg, metric, weights, terrain }: Args) {
  const query = useTierQuery(month, scope)
  const { search } = useLocation()
  const entries = useMemo<HeatEntry[]>(() => {
    const all = query.data?.items ?? []
    const items = scope === 'dong' ? all.filter((i) => i.sigungu?.id === sgg) : all
    const values = items.map((i) => Math.round(heatValue(i, metric, weights)))
    const grades = heatGrades(values)
    const current = new URLSearchParams(search)
    return items.map((item, i) => ({
      id: item.id,
      name: item.name,
      score: values[i],
      grade: grades[i],
      step: colorScale(values[i]),
      dimmed: scope === 'sigungu' && !terrainMatch(item, terrain),
      href: carryQuery(scope === 'dong' ? `/dongs/${item.id}` : `/regions/${item.id}`, current),
    }))
  }, [query.data, scope, sgg, metric, weights, terrain, search])
  return { entries, isPending: query.isPending, isError: query.isError, refetch: query.refetch }
}
```

```tsx
import { useCallback, useMemo } from 'react'
import { Map as GoogleMap } from '@vis.gl/react-google-maps'
import { useNavigate } from 'react-router-dom'
import PolygonLayer from '@/components/map/PolygonLayer'
import { bboxToBounds, toLatLngPaths } from '@/components/map/geo'
import type { GeoFeatureCollection, PolygonStyle } from '@/components/map/geo'
import { useGeoQuery } from '@/features/dong/useGeoQuery'
import { HEAT_VAR } from '@/features/heatmap/colorScale'
import type { HeatEntry } from '@/features/heatmap/useHeatEntries'
import { useSigunguGeoQuery } from '@/features/heatmap/useSigunguGeoQuery'
import type { TierScope } from '@/features/tier/types'

type Props = {
  scope: TierScope
  sgg: string | null
  /** 행정동 범위인데 시군구를 아직 고르지 않은 상태: 시군구 폴리곤을 회색으로 그리고 클릭하면 onPickSigungu */
  pickMode: boolean
  entries: HeatEntry[]
  hoverId: string | null
  onHover: (id: string | null) => void
  onPickSigungu: (id: string) => void
}

const DIM: PolygonStyle = {
  fill: 'var(--color-gray-300)',
  fillOpacity: 0.35,
  stroke: 'var(--color-gray-500)',
  strokeWidth: 1,
}

/** 폴리곤 전체를 감싸는 [west, south, east, north] */
function bboxOf(fc: GeoFeatureCollection): [number, number, number, number] {
  const pts = fc.features.flatMap((f) => toLatLngPaths(f.geometry).flat())
  const lngs = pts.map((p) => p.lng)
  const lats = pts.map((p) => p.lat)
  return [Math.min(...lngs), Math.min(...lats), Math.max(...lngs), Math.max(...lats)]
}

/**
 * Google Maps 위 폴리곤 히트맵. GoogleMapProvider(W-04) 안에서만 렌더된다.
 * 시군구 범위: sigungu.json 색칠, 클릭 → W-03. 행정동 범위: 고른 시군구의 동 폴리곤, 클릭 → W-04.
 * 색은 토큰 CSS 변수 문자열로 넘기고 PolygonLayer 가 resolveToken 으로 해석한다.
 */
export default function RegionHeatmap({
  scope,
  sgg,
  pickMode,
  entries,
  hoverId,
  onHover,
  onPickSigungu,
}: Props) {
  const navigate = useNavigate()
  const sigunguGeo = useSigunguGeoQuery()
  const dongGeo = useGeoQuery(scope === 'dong' && sgg ? sgg : undefined)
  const byId = useMemo(() => new Map(entries.map((e) => [e.id, e])), [entries])

  const getStyle = useCallback(
    (id: string): PolygonStyle => {
      const e = byId.get(id)
      const hovered = hoverId === id
      if (pickMode || !e || e.dimmed) return { ...DIM, strokeWidth: hovered ? 2 : 1 }
      return {
        fill: `var(${HEAT_VAR[e.step]})`,
        fillOpacity: hovered ? 0.85 : 0.65,
        stroke: 'var(--color-gray-800)',
        strokeWidth: hovered ? 2 : 1,
      }
    },
    [byId, hoverId, pickMode],
  )

  const getLabel = useCallback(
    (id: string) => {
      if (pickMode) return undefined
      const e = byId.get(id)
      return e ? `${e.name} ${e.score}` : undefined
    },
    [byId, pickMode],
  )

  const onClick = useCallback(
    (id: string) => {
      if (pickMode) {
        onPickSigungu(id)
        return
      }
      const e = byId.get(id)
      if (e) navigate(e.href)
    },
    [byId, navigate, onPickSigungu, pickMode],
  )

  const features = scope === 'dong' && !pickMode ? dongGeo.data : sigunguGeo.data
  if (!features || features.features.length === 0) return null

  // hover 툴팁 (스펙 §3-4). PolygonLayer 의 onHover 는 id 만 주므로 커서 위치 대신 지도 왼쪽 위에 고정한다.
  // hoverId 는 카드 hover 와 공유하므로 카드에 올려도 같은 툴팁이 뜬다(연동)
  const hovered = !pickMode && hoverId ? byId.get(hoverId) : undefined

  // PolygonLayer 는 useMap() 을 쓰므로 반드시 <Map> 안에 둔다. 범위(시군구/고른 시군구의 동)가 바뀌면 key 로 다시 맞춘다
  return (
    <div className="relative h-96 w-full overflow-hidden rounded-xl">
      {hovered && (
        <div
          role="tooltip"
          className="border-border-default bg-bg-surface shadow-float text-body-sm text-text-primary pointer-events-none absolute top-3 left-3 z-10 rounded-lg border px-3 py-2"
        >
          {hovered.name} · {hovered.score}점 · {hovered.grade}
        </div>
      )}
      <GoogleMap
        key={pickMode ? 'pick' : `${scope}:${sgg ?? ''}`}
        defaultBounds={bboxToBounds(bboxOf(features))}
        gestureHandling="cooperative"
        disableDefaultUI
        zoomControl
        style={{ width: '100%', height: '100%' }}
      >
        <PolygonLayer
          features={features}
          getStyle={getStyle}
          getLabel={getLabel}
          onHover={onHover}
          onClick={onClick}
        />
      </GoogleMap>
    </div>
  )
}
```

`Map`의 `style` 100%는 컨테이너 채우기용(W-04 `DongMap`과 같음). 지도 높이는 `h-96`(기본 spacing). 폴리곤 파일이 없는 시군구(`useGeoQuery`가 `EMPTY_GEO`)는 지도를 그리지 않는다. hover 툴팁은 지도 키가 있을 때만 보이므로 jsdom 테스트 대신 Task 11 수동 확인(`VITE_GOOGLE_MAPS_KEY` 설정 시 폴리곤 hover → 왼쪽 위 `양양군 · 54점 · {등급}`)으로 본다.
