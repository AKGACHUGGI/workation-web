# W-03 지역 상세 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** `/regions/:sggId`에서 시군구 하나의 점수·순위·등급(W-02와 같은 데이터·함수), 핵심 지표 5종, 월별 혼잡 막대, 주중·주말 차이, 규칙 기반 추천 사유, 행정동 목록, 커뮤니티(읽기), 지원사업 링크, sticky PlanCard를 보여준다.

**Architecture:** 점수·순위는 `GET /api/tier?month&scope=sigungu` 캐시를 재사용해 `rankIn(items, id, mode)`로 계산하고, 상세 수치는 `GET /api/regions/{id}?month`로 받는다. 문장·라벨은 `features/region/{copy,reason,format}.ts`의 순수 함수로 만들고 테스트한다. 차트는 Recharts `BarChart`(공유 설정 `crowdBarConfig.ts`, W-06 재사용). 쓰기 동작(즐겨찾기·팁·투표·사진)은 `useRequireLogin()` 스텁이 "로그인이 필요합니다" 토스트만 띄운다(W-07이 교체).

**Tech Stack:** React 19 · TypeScript 6 · Vite 8 · Tailwind v4 토큰 · react-router-dom 7 · @tanstack/react-query 5 · recharts 3 · lucide-react · Vitest 5 + Testing Library · Playwright

**Spec:** `docs/superpowers/specs/2026-11-07-W-03-지역상세-design.md`

```ts
import { NotFoundError } from '@/lib/errors'

// workation-backend 호출용 최소 fetch 래퍼.
const BASE_URL = import.meta.env.VITE_API_BASE_URL

export async function api<T>(path: string, init?: RequestInit): Promise<T> {
  const res = await fetch(`${BASE_URL}${path}`, {
    headers: { 'Content-Type': 'application/json', ...init?.headers },
    ...init,
  })
  if (res.status === 404) throw new NotFoundError(path)
  if (!res.ok) throw new Error(`API ${res.status} ${res.statusText}: ${path}`)
  return res.json() as Promise<T>
}
```

```ts
import type { Grade } from '@/features/metric/types'
import { buildTiers } from '@/features/tier/buildTiers'
import type { TierItem, TierMode } from '@/features/tier/types'

export type RankResult = {
  score: number // 반올림 전. 표시는 Math.round
  rank: number // 1부터. 같은 점수는 같은 순위
  total: number
  grade: Grade
  item: TierItem
}

/**
 * W-02 와 같은 buildTiers 로 등급을 매긴 뒤, S→C 순서로 펼쳐 id 의 순위를 찾는다.
 * id 가 items 에 없으면 null (마스킹·특수시설 제외 동).
 */
export function rankIn(items: TierItem[], id: string, mode: TierMode): RankResult | null {
  const entries = buildTiers(items, mode).flatMap((g) =>
    g.entries.map((e) => ({ ...e, grade: g.grade })),
  )
  const index = entries.findIndex((e) => e.item.id === id)
  if (index < 0) return null
  const mine = entries[index]
  const rank = entries.findIndex((e) => e.score === mine.score) + 1
  return { score: mine.score, rank, total: items.length, grade: mine.grade, item: mine.item }
}
```

```ts
/** W-02 와 같은 기본값 (직접 설정: 조용·시기·숙박) */
export const DEFAULT_CUSTOM_METRICS: MetricKey[] = ['quiet', 'crowd', 'overnight']

export type PresetMode = {
  mode: TierMode
  /** '조용한 곳' / '활기 있는 한 달' / '직접 설정 (조용 · 시기 · 숙박)' */
  presetLabel: string
  /** custom 일 때 고른 지표, 아니면 null */
  selected: MetricKey[] | null
}

/** preset·metrics 쿼리 → 밸류 탭 TierMode. W-03·W-04 PlanCard 가 공유한다. */
export function usePresetMode(): PresetMode {
  const [preset] = usePresetParam()
  const [metrics] = useQueryState<MetricKey[]>(
    'metrics',
    parseMetrics,
    DEFAULT_CUSTOM_METRICS,
    serializeMetrics,
  )
  if (preset === 'custom') {
    const short = metrics.map((k) => METRIC_SHORT_LABEL[k]).join(' · ')
    return {
      mode: { tab: 'value', weights: customWeights(metrics), selected: metrics },
      presetLabel: `${PRESET_LABEL.custom} (${short})`,
      selected: metrics,
    }
  }
  return {
    mode: { tab: 'value', weights: PRESET_WEIGHTS[preset] },
    presetLabel: PRESET_LABEL[preset],
    selected: null,
  }
}
```

```ts
export function buildReason(i: ReasonInput): string[] {

  const inflow = i.dayInflow.toFixed(2)

  const s1 = `${i.monthLabel} ${i.presetLabel} 프리셋 종합 ${i.composite}점으로 ${i.total}곳 중 ${i.rank}위(${i.grade})입니다.`

        : `숙박 계수 ${coef}로 당일 방문 비중이 큰 지역이고,`

        : `낮 유입 지수 ${inflow}로 낮에 사람이 몰리는 편입니다.`
```

```ts
import type { MonthParam } from '@/features/metric/month'
import { monthNumber } from '@/features/metric/month'
import type { RegionDetailResponse, RegionDong } from '@/features/region/types'
import { SIGUNGU_MANUAL } from '@/mocks/region/manual'

/** 문자열 → 0~1 결정적 난수 (같은 id·month 는 항상 같은 값) */
function seeded(key: string): () => number {
  let h = 2166136261
  for (const ch of key) h = Math.imul(h ^ ch.charCodeAt(0), 16777619)
  return () => {
    h = Math.imul(h ^ (h >>> 15), 2246822507)
    h = Math.imul(h ^ (h >>> 13), 3266489909)
    return ((h ^= h >>> 16) >>> 0) / 4294967296
  }
}

const SEASON_BASE: Record<string, number[]> = {
  summer_beach: [40, 35, 40, 50, 60, 70, 95, 100, 60, 55, 45, 42],
  spring_autumn: [45, 48, 70, 95, 85, 60, 55, 60, 70, 100, 80, 50],
  second_half_rise: [53, 41, 41, 46, 60, 60, 80, 100, 52, 68, 52, 47],
  flat: [80, 78, 85, 90, 92, 88, 95, 100, 90, 92, 85, 82],
}

function topN(values: number[], n: number, dir: 'asc' | 'desc'): number[] {
  return values
    .map((v, i) => ({ v, m: i + 1 }))
    .sort((a, b) => (dir === 'asc' ? a.v - b.v : b.v - a.v))
    .slice(0, n)
    .map((x) => x.m)
    .sort((a, b) => a - b)
}

/**
 * 수작업 픽스처가 없는 시군구·월 조합용. scripts/gen-mock.py 가 실제 CSV 로 대체할 때까지 쓴다.
 * 월 간 일관성: stats 외 값(months·dongs 이름)은 month 와 무관하게 id 만으로 결정한다.
 */
export function makeRegionMock(id: string, month: MonthParam): RegionDetailResponse {
  const m = SIGUNGU_MANUAL[id]
  const rnd = seeded(id)
  const base = SEASON_BASE[m.seasonType]
  const months = base.map((crowd, i) => ({
    month: i + 1,
    crowd: Math.min(100, Math.round(crowd + rnd() * 6 - 3)),
  }))
  const max = Math.max(...months.map((x) => x.crowd))
  const norm = months.map((x) => ({ month: x.month, crowd: Math.round((x.crowd / max) * 100) }))
  const n = monthNumber(month)
  const crowdIndex =
    n === null ? Math.round(norm.reduce((s, x) => s + x.crowd, 0) / 12) : norm[n - 1].crowd

  const dongNames = ['중앙동', '동부면', '서부면', '남부면', '북부면']
  const dongs: RegionDong[] = dongNames.map((name, i) => {
    const peakMonth = topN(
      norm.map((x) => x.crowd),
      1,
      'desc',
    )[0]
    const quietScore = Math.round(15 + rnd() * 70)
    return {
      id: `${id}${String(25000 + i * 1000).padStart(5, '0')}`,
      name,
      kind: i % 2 === 0 ? 'living' : 'tourist',
      landmark: `${m.name} ${name} 중심지`,
      quietScore,
      peakMonth,
      crowdOfPeak: n === null ? 100 : Math.max(20, Math.round(norm[n - 1].crowd - rnd() * 15)),
    }
  })

  return {
    id,
    name: m.name,
    province: m.province,
    tags: m.tags,
    seasonType: m.seasonType,
    photos: m.photos,
    month,
    stats: {
      crowdIndex,
      dayInflow: Number((0.6 + rnd() * 0.7).toFixed(2)),
      quietRank: 1 + Math.floor(rnd() * 12),
      peerShare: Math.round(30 + rnd() * 25),
      overnightCoef: Number((0.5 + rnd() * 0.5).toFixed(2)),
      weekendRatio: Number((0.85 + rnd() * 0.45).toFixed(2)),
    },
    months: norm,
    bestMonths: topN(
      norm.map((x) => x.crowd),
      2,
      'asc',
    ),
    avoidMonths: topN(
      norm.map((x) => x.crowd),
      2,
      'desc',
    ),
    peerOffpeakMonths: topN(
      norm.map((x) => x.crowd),
      2,
      'asc',
    ),
    dongs: [...dongs].sort((a, b) => b.quietScore - a.quietScore),
    excludedDongs: [],
    links: m.links,
  }
}
```

```tsx
import { useCallback, useEffect, useMemo, useRef, useState } from 'react'
import type { ReactNode } from 'react'
import Toast from '@/components/ui/toast/Toast'
import { ToastContext } from '@/components/ui/toast/toastContext'
import type { ToastApi, ToastOptions, ToastState } from '@/components/ui/toast/toastContext'

/** 한 번에 1개. 새 show 는 이전 토스트를 대체한다. RootLayout 이 감싼다. */
export default function ToastProvider({ children }: { children: ReactNode }) {
  const [toast, setToast] = useState<ToastState | null>(null)
  const timer = useRef<ReturnType<typeof setTimeout> | null>(null)
  const seq = useRef(0)

  const dismiss = useCallback(() => {
    if (timer.current) clearTimeout(timer.current)
    timer.current = null
    setToast(null)
  }, [])

  const show = useCallback((message: string, opts?: ToastOptions) => {
    if (timer.current) clearTimeout(timer.current)
    const id = ++seq.current
    setToast({ id, message, action: opts?.action })
    const ms = opts?.durationMs ?? (opts?.action ? 5000 : 3000)
    timer.current = setTimeout(() => {
      setToast((cur) => (cur?.id === id ? null : cur))
      timer.current = null
    }, ms)
  }, [])

  useEffect(
    () => () => {
      if (timer.current) clearTimeout(timer.current)
    },
    [],
  )

  const api = useMemo<ToastApi>(() => ({ show, dismiss }), [show, dismiss])

  return (
    <ToastContext.Provider value={api}>
      {children}
      {toast && (
        <Toast key={toast.id} message={toast.message} action={toast.action} onDismiss={dismiss} />
      )}
    </ToastContext.Provider>
  )
}
```

```ts
/** '3일 전', '2개월 전', '방금' */
export function relativeTime(iso: string, now: Date = new Date()): string {
  const diff = Math.max(0, now.getTime() - new Date(iso).getTime())
  const day = 24 * 60 * 60 * 1000
  if (diff < 60 * 60 * 1000) return '방금'
  if (diff < day) return `${Math.floor(diff / (60 * 60 * 1000))}시간 전`
  if (diff < 30 * day) return `${Math.floor(diff / day)}일 전`
  if (diff < 365 * day) return `${Math.floor(diff / (30 * day))}개월 전`
  return `${Math.floor(diff / (365 * day))}년 전`
}
```
