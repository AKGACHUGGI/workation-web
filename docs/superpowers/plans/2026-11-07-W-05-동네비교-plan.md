- [ ] **Step 4: useFocusTrap 구현**

`src/hooks/useFocusTrap.ts`:

```ts
import { useEffect, useRef } from 'react'
import type { RefObject } from 'react'

const FOCUSABLE =
  'a[href], button:not([disabled]), input:not([disabled]), select:not([disabled]), textarea:not([disabled]), [tabindex]:not([tabindex="-1"])'

/**
 * active 동안 ref 안에 포커스를 가둔다.
 * - 활성화 시 첫 포커스 가능 요소(없으면 컨테이너)로 포커스
 * - Tab / Shift+Tab 순환, Esc → onEscape
 * - 비활성화(또는 언마운트) 시 활성화 직전에 포커스가 있던 요소로 복귀
 */
export function useFocusTrap(
  ref: RefObject<HTMLElement | null>,
  active: boolean,
  onEscape?: () => void,
) {
  const onEscapeRef = useRef(onEscape)
  onEscapeRef.current = onEscape

  useEffect(() => {
    if (!active) return
    const container = ref.current
    if (!container) return
    const opener = document.activeElement instanceof HTMLElement ? document.activeElement : null
    const focusables = () => Array.from(container.querySelectorAll<HTMLElement>(FOCUSABLE))

    ;(focusables()[0] ?? container).focus()

    const onKeyDown = (e: KeyboardEvent) => {
      if (e.key === 'Escape') {
        e.preventDefault()
        onEscapeRef.current?.()
        return
      }
      if (e.key !== 'Tab') return
      const items = focusables()
      if (items.length === 0) {
        e.preventDefault()
        return
      }
      const first = items[0]
      const last = items[items.length - 1]
      const current = document.activeElement
      if (e.shiftKey && (current === first || !container.contains(current))) {
        e.preventDefault()
        last.focus()
      } else if (!e.shiftKey && (current === last || !container.contains(current))) {
        e.preventDefault()
        first.focus()
      }
    }
    document.addEventListener('keydown', onKeyDown)
    return () => {
      document.removeEventListener('keydown', onKeyDown)
      opener?.focus()
    }
  }, [ref, active])
}
```

`src/components/ui/Modal.tsx`:

```tsx
import { useEffect, useRef } from 'react'
import type { ReactNode } from 'react'
import { createPortal } from 'react-dom'
import { useFocusTrap } from '@/hooks/useFocusTrap'

type Props = {
  open: boolean
  onClose: () => void
  /** 제목 요소의 id (aria-labelledby) */
  titleId: string
  children: ReactNode
  /** sm 440px (로그인), md 640px (동 추가) */
  width?: 'sm' | 'md'
}

/** 공통 모달. 스크림(scrim 토큰) 위 가운데, 포커스 트랩, Esc·스크림 클릭 닫기, body 스크롤 잠금 */
export default function Modal({ open, onClose, titleId, children, width = 'md' }: Props) {
  const ref = useRef<HTMLDivElement>(null)
  useFocusTrap(ref, open, onClose)

  useEffect(() => {
    if (!open) return
    const prev = document.body.style.overflow
    document.body.style.overflow = 'hidden'
    return () => {
      document.body.style.overflow = prev
    }
  }, [open])

  if (!open) return null

  return createPortal(
    <div
      data-testid="modal-scrim"
      className="bg-scrim/50 fixed inset-0 z-50 flex items-center justify-center py-8"
      onMouseDown={(e) => {
        if (e.target === e.currentTarget) onClose()
      }}
    >
      <div
        ref={ref}
        role="dialog"
        aria-modal="true"
        aria-labelledby={titleId}
        tabIndex={-1}
        className={`bg-bg-surface shadow-float max-h-full overflow-y-auto rounded-2xl p-6 ${width === 'sm' ? 'w-modal-sm' : 'w-modal-md'}`}
      >
        {children}
      </div>
    </div>,
    document.body,
  )
}
```

높이 제한은 임의값(`max-h-[90vh]`) 대신 스크림 `py-8`(위아래 32px 여백) + 패널 `max-h-full`로 건다. 스크림이 `fixed inset-0`이라 높이가 확정되므로 `max-h-full`(100%)은 "뷰포트 − 64px"로 계산되고, 넘치면 패널 안에서 `overflow-y-auto`로 스크롤된다. Tailwind v4 기본 유틸에는 `max-h-screen`(100vh)·`max-h-dvh`만 있고 90vh 유틸은 없으므로 이 방식이 토큰 추가 없이 끝난다.

```ts
export const MAX_COMPARE = 3

/**
 * `dongs=` 쿼리 파싱. 순서 유지 · 중복 제거 · validIds 에 없는 id 는 dropped · 앞 MAX_COMPARE 개.
 * validIds 가 없으면(티어 응답 전) 검증을 건너뛴다.
 */
export function parseDongsParam(
  raw: string | null,
  validIds?: ReadonlySet<string>,
): { ids: string[]; dropped: string[] } {
  if (!raw) return { ids: [], dropped: [] }
  const unique = Array.from(new Set(raw.split(',').filter(Boolean)))
  const dropped: string[] = []
  const kept: string[] = []
  for (const id of unique) {
    if (validIds && !validIds.has(id)) dropped.push(id)
    else kept.push(id)
  }
  return { ids: kept.slice(0, MAX_COMPARE), dropped }
}
```

```ts
export const COMPARE_STORAGE_KEY = 'workation.compare.dongs'

/** 마지막 비교 목록. 없거나 깨졌으면 [] */
export function readStoredDongs(): string[] {
  try {
    const raw = localStorage.getItem(COMPARE_STORAGE_KEY)
    if (!raw) return []
    const parsed: unknown = JSON.parse(raw)
    return Array.isArray(parsed) ? parsed.filter((v): v is string => typeof v === 'string') : []
  } catch {
    return []
  }
}

/** 실패(용량·프라이빗 모드)는 조용히 무시 */
export function writeStoredDongs(ids: string[]): void {
  try {
    localStorage.setItem(COMPARE_STORAGE_KEY, JSON.stringify(ids))
  } catch {
    /* ignore */
  }
}
```

```ts
import type { TierItem } from '@/features/tier/types'

/** 같은 시군구 행정동 중 동네 조용함(scores.quiet) 순위. 동점은 같은 순위. 없으면 null */
export function rankInSigungu(
  items: TierItem[],
  dongId: string,
): { rank: number; total: number } | null {
  const me = items.find((i) => i.id === dongId)
  if (!me?.sigungu) return null
  const siblings = items.filter((i) => i.sigungu?.id === me.sigungu!.id)
  const higher = siblings.filter((i) => i.scores.quiet > me.scores.quiet).length
  return { rank: higher + 1, total: siblings.length }
}
```

```ts
import { useCallback, useEffect, useMemo } from 'react'
import { useSearchParams } from 'react-router-dom'
import { MAX_COMPARE, parseDongsParam } from '@/features/compare/params'
import { readStoredDongs, writeStoredDongs } from '@/features/compare/storage'

const KEY = 'dongs'

/**
 * 비교 목록. URL `dongs` 가 진실, localStorage 는 복원·백업.
 * validIds: tier dong 응답의 id 목록 (메모이즈해서 넘길 것). 없으면 검증 생략.
 */
export function useCompareList(validIds?: readonly string[]) {
  const [searchParams, setSearchParams] = useSearchParams()
  const raw = searchParams.get(KEY)

  const validSet = useMemo(() => (validIds ? new Set(validIds) : undefined), [validIds])
  const { ids, dropped } = useMemo(() => parseDongsParam(raw, validSet), [raw, validSet])
  const serialized = ids.join(',')

  const write = useCallback(
    (next: string[], replace: boolean) => {
      setSearchParams(
        (prev) => {
          const sp = new URLSearchParams(prev)
          if (next.length === 0) sp.delete(KEY)
          else sp.set(KEY, next.join(','))
          return sp
        },
        { replace },
      )
    },
    [setSearchParams],
  )

  // 1) 쿼리 없음 → 마지막 목록 복원 (replace)
  useEffect(() => {
    if (raw !== null) return
    const stored = readStoredDongs()
    if (stored.length > 0) write(stored, true)
  }, [raw, write])

  // 2) 파싱 결과가 raw 와 다르면(중복·초과·무효) URL 보정 (replace)
  useEffect(() => {
    if (raw === null || raw === serialized) return
    write(ids, true)
  }, [raw, serialized, ids, write])

  // 3) 목록이 바뀌면 storage 갱신 (URL 변경 후)
  useEffect(() => {
    if (raw !== null) writeStoredDongs(ids)
  }, [raw, ids])

  const canAdd = ids.length < MAX_COMPARE

  const add = useCallback(
    (id: string) => {
      if (!canAdd || ids.includes(id)) return
      write([...ids, id], false)
    },
    [canAdd, ids, write],
  )

  const remove = useCallback(
    (id: string) => {
      const next = ids.filter((v) => v !== id)
      if (next.length === 0) writeStoredDongs([]) // 복원 효과가 되살리지 않도록 먼저 비운다
      write(next, false)
    },
    [ids, write],
  )

  return { ids, add, remove, canAdd, dropped }
}
```

```tsx
  const [query, setQuery] = useState('')
  const [sgg, setSgg] = useState<string>(ALL)

    return items
```

```tsx
  const validIds = useMemo(() => tierItems?.map((i) => i.id), [tierItems])

  }, [droppedKey, toast])

  const details = useQueries({
    queries: ids.map((id) => ({
      queryKey: dongKeys.detail(id, month),
      queryFn: () => fetchDong(id, month),
      placeholderData: keepPreviousData,
    })),
  })

  const entries = ids.map((id, i) => {
    const d = details[i].data
    const t = tierItems?.find((it) => it.id === id)
    return {
      id,
      name: d?.name ?? t?.name ?? id,
      sigunguName: d?.sigungu.name ?? t?.sigungu?.name ?? '',
    }
```
