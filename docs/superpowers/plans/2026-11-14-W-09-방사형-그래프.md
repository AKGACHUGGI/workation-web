```ts
import { SIGUNGU_NODE, typeNodeId } from '@/features/explore/sigunguGraph'
import { TYPE_NAMES } from '@/features/explore/types'
import type { PinDong, TypeName } from '@/features/explore/types'

/** 동 노드 반지름. 점수 비례 크기는 뜻이 잘 안 읽혀 모두 같게 둔다 (2026-11-13 사용자 결정) */
export const DONG_R = 9
/** 유형 노드 반지름 (v6 사실 노드 52×52 보다 조금 크게, 5글자 "네트워킹형"이 들어간다) */
export const TYPE_R = 30
/** 가운데 시군구 노드 반지름 (유형보다 크게) */
export const SIGUNGU_R = 34
/** 같은 링 이웃 동 중심 사이 최소 거리 = 동 지름 + 이름 여백 */
export const MIN_GAP = 2 * DONG_R + 22

/** 반지름 = 그래프 영역 짧은 변 × 비율 (스펙 §4-6) */
const RING = { type: 0.17, inner: 0.3, outer: 0.42 } as const

export type Size = { w: number; h: number }
export type Point = { x: number; y: number; r: number }

const rad = (deg: number) => (deg * Math.PI) / 180

function ringRadius(size: Size) {
  const s = Math.min(size.w, size.h)
  return { type: s * RING.type, inner: s * RING.inner, outer: s * RING.outer }
}

/** n개 동이 바깥 링 한 줄에 들어가지 않으면 true (동은 전체를 같은 간격 360/n 으로 나눠 앉는다) */
export function twoRings(n: number, size: Size): boolean {
  return n > 0 && ringRadius(size).outer * rad(360 / n) < MIN_GAP
}

/** 전체 i번째 동 자리. 각도 = startDeg + (i + 0.5) × 360/n. 두 줄이면 홀수 번째는 안쪽 링 */
export function dongSlot(i: number, n: number, startDeg: number, size: Size): Point {
  const ring = ringRadius(size)
  const r = twoRings(n, size) && i % 2 === 1 ? ring.inner : ring.outer
  const a = rad(startDeg + ((i + 0.5) * 360) / n)
  return { x: size.w / 2 + r * Math.cos(a), y: size.h / 2 + r * Math.sin(a), r: DONG_R }
}

/** 0..n-1 을 가운데부터 바깥으로 번갈아 (n=4 → [1, 2, 0, 3]) */
export function centerOut(n: number): number[] {
  const mid = (n - 1) / 2
  return [...Array(n).keys()].sort((a, b) => Math.abs(a - mid) - Math.abs(b - mid) || a - b)
}

const byName = (a: PinDong, b: PinDong) => a.name.localeCompare(b.name, 'ko')

/**
 * ① 방사형 배치 (스펙 §4). 가운데 시군구, 안쪽 링 유형, 바깥 링 동.
 * 각도 몫 = 동 수 비례. 유형 없음 무리는 가운데가 아래(90°), 유형은 이어서 시계 방향 몰입 → 네트워킹 → 로컬.
 * 무리 안: 유형은 점수 높은 순으로 가운데부터 번갈아, 유형 없음은 이름 가나다순
 */
export function radialLayout(pins: PinDong[], size: Size): Map<string, Point> {
  const out = new Map<string, Point>()
  const cx = size.w / 2
  const cy = size.h / 2
  out.set(SIGUNGU_NODE, { x: cx, y: cy, r: SIGUNGU_R })
  const n = pins.length
  if (n === 0) return out

  const step = 360 / n
  const none = pins.filter((p) => p.topType === null).sort(byName)
  const groups: { type: TypeName | null; dongs: PinDong[] }[] = [
    { type: null, dongs: none },
    ...TYPE_NAMES.map((t) => ({
      type: t,
      dongs: pins
        .filter((p) => p.topType === t)
        .sort((a, b) => (b.topScore ?? 0) - (a.topScore ?? 0) || byName(a, b)),
    })),
  ].filter((g) => g.dongs.length > 0)

  const start = 90 - (none.length * step) / 2
  const typeR = ringRadius(size).type
  let i = 0
  for (const g of groups) {
    const k = g.dongs.length
    const order = g.type === null ? [...Array(k).keys()] : centerOut(k)
    g.dongs.forEach((p, rank) => out.set(p.id, dongSlot(i + order[rank], n, start, size)))
    if (g.type !== null) {
      const mid = rad(start + (i + k / 2) * step)
      out.set(typeNodeId(g.type), {
        x: cx + typeR * Math.cos(mid),
        y: cy + typeR * Math.sin(mid),
        r: TYPE_R,
      })
    }
    i += k
  }
  return out
}
```

```ts
/** 실측이 없는 동: id 해시로 정한 시안용 값. 1위 20~94, 백엔드처럼 TYPE_MIN 미만이면 유형 없음 */
function seedPin(id: string, name: string, center: LatLng, tags: string[]): PinDong {
  const h = hashString(id)
  const best = 20 + (h % 75)
  const typed = best >= TYPE_MIN
  return {
    id,
    name,
    center,
    tags,
    topType: typed ? TYPE_NAMES[h % TYPE_NAMES.length] : null,
    topScore: typed ? best : null,
    coastal: tags.includes('coast'),
    life: h % 3 === 0 ? '관광지' : '생활권',
    grades: { quiet: GRADES[(h >>> 3) % 4], evening: GRADES[(h >>> 6) % 4] },
  }
}
```

```ts
import { TYPE_NAMES } from '@/features/explore/types'
import type { PinDong, TypeName } from '@/features/explore/types'

/** ① 가운데 시군구 노드 id (방사형, 2026-11-14) */
export const SIGUNGU_NODE = 'sigungu'
/** ① 유형 노드 id */
export const typeNodeId = (t: TypeName) => `type:${t}`

export type GraphNode =
  | { id: string; kind: 'sigungu'; label: string }
  | { id: string; kind: 'type'; label: TypeName }
  | { id: string; kind: 'dong'; label: string; type: TypeName | null; coastal: boolean }
/** dashed = 시군구 ─ 대표 유형 없는 동 */
export type GraphLink = { source: string; target: string; dashed: boolean }
export type SigunguGraphData = { nodes: GraphNode[]; links: GraphLink[] }

/**
 * ① 방사형 지식그래프 (스펙 2026-11-14 §3). 노드 = 시군구 1 · 유형(이어진 동이 있는 것만) · 동.
 * 선 = 시군구─유형, 유형─동, 시군구─유형 없는 동(점선). 노드 순서 = 그리는 순서 = Tab 순서(유형 → 동)
 */
export function buildSigunguGraph(name: string, pins: PinDong[]): SigunguGraphData {
  const types = TYPE_NAMES.filter((t) => pins.some((p) => p.topType === t))
  const nodes: GraphNode[] = [
    { id: SIGUNGU_NODE, kind: 'sigungu', label: name },
    ...types.map((t): GraphNode => ({ id: typeNodeId(t), kind: 'type', label: t })),
    ...pins.map((p): GraphNode => ({
      id: p.id,
      kind: 'dong',
      label: p.name,
      type: p.topType,
      coastal: p.coastal,
    })),
  ]
  const links: GraphLink[] = [
    ...types.map((t) => ({ source: SIGUNGU_NODE, target: typeNodeId(t), dashed: false })),
    ...pins.map((p) =>
      p.topType
        ? { source: typeNodeId(p.topType), target: p.id, dashed: false }
        : { source: SIGUNGU_NODE, target: p.id, dashed: true },
    ),
  ]
  return { nodes, links }
}

/** 선택 카드 사실 칩과 공통 비교에 쓰는 핀 사실 (스펙 §5): 대표 유형 · 바닷가 · 생활권/관광지 · 조용함 · 저녁 활기 */
export function pinFacts(pin: PinDong): string[] {
  return [
    pin.topType,
    pin.coastal ? '바닷가' : null,
    pin.life,
    pin.grades.quiet && `조용함 ${pin.grades.quiet}`,
    pin.grades.evening && `저녁 활기 ${pin.grades.evening}`,
  ].filter((f): f is string => Boolean(f))
}

/**
 * 선택 카드의 `공통 n: …` (스펙 §5). 그 동과 핀 사실이 많이 겹치는 동 순, 같으면 이름 순, 0개 제외.
 * common 은 겹치는 사실 (pinFacts 순)
 */
export function sharedFacts(
  pins: PinDong[],
  dongId: string,
): { dongId: string; name: string; common: string[] }[] {
  const me = pins.find((p) => p.id === dongId)
  if (!me) return []
  const mine = pinFacts(me)
  return pins
    .filter((p) => p.id !== dongId)
    .map((p) => {
      const theirs = new Set(pinFacts(p))
      return { dongId: p.id, name: p.name, common: mine.filter((f) => theirs.has(f)) }
    })
    .filter((s) => s.common.length > 0)
    .sort((a, b) => b.common.length - a.common.length || a.name.localeCompare(b.name, 'ko'))
}

/** 유형 노드 클릭 → 그 유형이 대표인 동 (핀 순서) */
export function typeDongs(pins: PinDong[], type: TypeName): PinDong[] {
  return pins.filter((p) => p.topType === type)
}
```

```tsx
import type { KeyboardEvent, MouseEvent } from 'react'
import { EXPLORE_COPY } from '@/features/explore/copy'
import type { Point } from '@/features/explore/radialLayout'
import type { GraphNode, SigunguGraphData } from '@/features/explore/sigunguGraph'
import { GRAPH_H, GRAPH_W } from '@/features/explore/components/panel/panelLayout'
import { NODE_FILL } from '@/features/explore/components/panel/relationColor'

export type GraphPick = { kind: 'dong' | 'type'; id: string }

type Props = {
  name: string
  graph: SigunguGraphData
  /** radialLayout 결과 (노드 id → 자리) */
  layout: Map<string, Point>
  pick: GraphPick | null
  onPick: (pick: GraphPick | null) => void
  hoveredDong: string | null
  onHoverDong: (dongId: string | null) => void
}

/** 강조 대상 노드 id (그 노드 + 선으로 이어진 이웃). focus 가 없으면 null = 모두 진하게 */
function litIds(graph: SigunguGraphData, focusId: string | null): Set<string> | null {
  if (!focusId) return null
  const ids = new Set([focusId])
  for (const l of graph.links) {
    if (l.source === focusId) ids.add(l.target)
    if (l.target === focusId) ids.add(l.source)
  }
  return ids
}

const DIM = 0.18

/**
 * ① 시군구 지식그래프, 방사형 (스펙 2026-11-14 §3·§5). 가운데 시군구 → 유형 → 동, 유형 없는 동은 점선.
 * 자리는 radialLayout 이 정하고 여기서는 그리기만 한다 (끌기 없음). 클릭은 그래프 안 강조만(서버 요청 없음).
 * 시군구 노드 클릭 = 빈 곳 클릭 = 선택 해제. 지도 핀 hover 는 hoveredDong 으로 받아 미리 강조
 */
export default function SigunguGraph({
  name,
  graph,
  layout,
  pick,
  onPick,
  hoveredDong,
  onHoverDong,
}: Props) {
  const focusId = hoveredDong ?? pick?.id ?? null
  const lit = litIds(graph, focusId)

  const pickOf = (n: GraphNode): GraphPick | null =>
    n.kind === 'sigungu' ? null : { kind: n.kind, id: n.id }
  const onNodeClick = (n: GraphNode) => (e: MouseEvent) => {
    e.stopPropagation()
    onPick(pickOf(n))
  }
  const onNodeKey = (n: GraphNode) => (e: KeyboardEvent) => {
    if (e.key !== 'Enter' && e.key !== ' ') return
    e.preventDefault()
    onPick(pickOf(n))
  }

  return (
    <svg
      role="group"
      aria-label={EXPLORE_COPY.graphLabel(name)}
      viewBox={`0 0 ${GRAPH_W} ${GRAPH_H}`}
      className="h-100 w-full shrink-0 select-none"
      onClick={() => onPick(null)}
    >
      <g>
        {graph.links.map((l) => {
          const a = layout.get(l.source)
          const b = layout.get(l.target)
          if (!a || !b) return null
          const on = focusId === null || l.source === focusId || l.target === focusId
          const strong = focusId !== null && on
          return (
            <line
              key={`${l.source}|${l.target}`}
              x1={a.x}
              y1={a.y}
              x2={b.x}
              y2={b.y}
              data-dashed={l.dashed || undefined}
              strokeDasharray={l.dashed ? '3 3' : undefined}
              className={strong ? 'stroke-text-primary' : 'stroke-border-strong'}
              strokeWidth={strong ? 1.5 : 1}
              opacity={on ? 1 : DIM}
            />
          )
        })}
      </g>
      {graph.nodes.map((n) => {
        const p = layout.get(n.id)
        if (!p) return null
        const on = lit === null || lit.has(n.id)
        if (n.kind === 'sigungu') {
          return (
            <g
              key={n.id}
              data-kind="sigungu"
              transform={`translate(${p.x} ${p.y})`}
              className="cursor-pointer"
              onClick={onNodeClick(n)}
            >
              <circle r={p.r} fill={NODE_FILL.sigungu} />
              <text
                textAnchor="middle"
                dominantBaseline="central"
                className="fill-text-inverse text-explore-xs font-bold"
              >
                {n.label}
              </text>
            </g>
          )
        }
        const selected = focusId === n.id
        return (
          <g
            key={n.id}
            role="button"
            tabIndex={0}
            aria-label={n.label}
            aria-pressed={pick?.id === n.id}
            data-kind={n.kind}
            data-dim={on ? undefined : 'true'}
            opacity={on ? 1 : DIM}
            transform={`translate(${p.x} ${p.y})`}
            className="cursor-pointer outline-none"
            onClick={onNodeClick(n)}
            onKeyDown={onNodeKey(n)}
            onPointerEnter={n.kind === 'dong' ? () => onHoverDong(n.id) : undefined}
            onPointerLeave={n.kind === 'dong' ? () => onHoverDong(null) : undefined}
          >
            {n.kind === 'dong' ? (
              <>
                <circle
                  r={p.r}
                  className="fill-bg-surface stroke-brand-primary"
                  strokeWidth={selected ? 3.5 : 2.5}
                />
                {n.coastal && (
                  <text data-coast aria-hidden x={p.r * 0.6} y={-p.r * 0.6} fontSize={10}>
                    🌊
                  </text>
                )}
                <text
                  y={p.r + 10}
                  textAnchor="middle"
                  className={`text-explore-3xs ${selected ? 'fill-text-primary font-bold' : 'fill-text-tertiary'}`}
                >
                  {n.label}
                </text>
              </>
            ) : (
              <>
                <circle r={p.r} fill={NODE_FILL.type} />
                <text
                  textAnchor="middle"
                  dominantBaseline="central"
                  className="fill-text-inverse text-explore-2xs font-bold"
                >
                  {n.label}
                </text>
              </>
            )}
          </g>
        )
      })}
    </svg>
  )
}
```
