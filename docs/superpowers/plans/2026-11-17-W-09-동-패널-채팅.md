`src/features/interest/chatInterest.ts`:

```ts
import { buildProfile } from '@/features/interest/profile'
import { getEvents } from '@/features/interest/store'
import type { ChatInterest, InterestProfile } from '@/features/interest/types'

const DONG_CODE = /^\d{10}$/
const RECENT_MAX = 3

/** 0 보다 큰 칸만 남긴다 */
function positive<K extends string>(o: Record<K, number>): Partial<Record<K, number>> {
  return Object.fromEntries(Object.entries<number>(o).filter(([, v]) => v > 0)) as Partial<
    Record<K, number>
  >
}

/** 관심 프로필 → 채팅 요청용 관심 기록. 시군구 5자리 코드는 recent 에서 뺀다 */
export function toChatInterest(p: InterestProfile): ChatInterest {
  return {
    metric: positive(p.metric),
    terrain: positive(p.terrain),
    recent: Object.entries(p.place)
      .filter(([code]) => DONG_CODE.test(code))
      .sort((a, b) => b[1] - a[1])
      .slice(0, RECENT_MAX)
      .map(([code]) => code),
  }
}

/** 지금 브라우저 기록으로 만든 채팅 요청용 관심 기록. 기록이 없으면 빈 값 */
export function chatInterest(): ChatInterest {
  return toChatInterest(buildProfile(getEvents(), Date.now()))
}
```

```ts
/** POST /api/explore/chat 요청. chips 는 웹에서 늘 [] (웹 스펙 §2) */
export type ChatRequest = {
  conversationId: string | null
  panel: string
  question: string
  criteria: { chips: string[]; interest: ChatInterest }
}
/**
 * POST /api/explore/chat 응답. 모르는 질문·다른 시군구·시군구 이름만 말함은 sentences 가 비고 guide 만,
 * 동 4개 이상이면 guide 와 비교 sentences 가 함께 온다
 */
export type ChatResponse = {
  conversationId: string
  dongs: { id: string; name: string }[]
  intent: string
  sentences: ChatSentence[]
  guide: string | null
  examples: string[]
}
/** retry = 네트워크·503 등(다시 시도), limit = 429, invalid = 400 */
export type ChatErrorKind = 'retry' | 'limit' | 'invalid'
export type ChatTurn =
  | { id: string; question: string; status: 'pending' }
  | {
      id: string
      question: string
      status: 'done'
      sentences: ChatSentence[]
      guide: string | null
    }
  | { id: string; question: string; status: 'error'; kind: ChatErrorKind }
/** 동 하나의 대화. 화면이 떠 있는 동안만 QueryClient 에 남는다. conversationId 는 서버 대화 기억 id */
export type ChatThread = { conversationId: string | null; turns: ChatTurn[]; examples: string[] }
```

```ts
  examples: [CANONICAL.weekday_hours, CANONICAL.evening, CANONICAL.quiet],
}
const GUIDE_OTHER = {
  intent: 'unknown_other',
  text: `${SEES} 그 질문엔 아직 답하기 어려워요. 이런 건 알려 드릴 수 있어요.`,
  examples: [CANONICAL.quiet, CANONICAL.months, CANONICAL.similar],
}
/** 백엔드 explore/chat/resolve.py SAME_SGG_ONLY */
const SAME_SGG_ONLY = '지금은 같은 시군구 안의 동네끼리만 비교할 수 있어요.'
/** 백엔드 explore/chat/flow.py COMPARE_EXAMPLES */
const COMPARE_EXAMPLES = ['둘이 비슷해?', '주말엔 몇 시가 붐벼?', '언제 가면 한산해?']
const NO_DATA = '이 동네는 그 내용을 계산할 데이터가 부족해요.'
const GRADE_NAME = { quiet: '조용함', peer: '또래 밀도', evening: '저녁 활기' } as const

/** 백엔드 규칙 문장을 동 상세 값으로 만든다. 전국 순위는 동 상세에 없어 뺀다 */
function ruleSentence(intent: Intent, d: DongDetail): string {
  switch (intent) {
    case 'weekday_hours':
    case 'weekend_hours': {
      const day = intent === 'weekday_hours' ? 'weekday' : 'weekend'
      const { peakHour, quietHour } = d.hourly[day]
      if (peakHour === null || quietHour === null) return NO_DATA
      return day === 'weekday'
        ? `평일 차량은 ${peakHour}시에 가장 많고, 업무시간엔 ${quietHour}시가 가장 한산해요.`
        : `주말 차량은 ${peakHour}시에 가장 많고, ${quietHour}시가 가장 한산해요.`
    }
    case 'months': {
      const values = d.months.raw.map((v) => v ?? -Infinity)
      const busiest = values.indexOf(Math.max(...values)) + 1
      return `${topicOf(d.months.basis)} ${d.months.quiet.join('·')}월이 가장 한산하고 ${busiest}월이 가장 붐벼요.`
    }
    case 'quiet':
    case 'peer':
    case 'evening': {
      const g = d.grades[intent]
      return g ? `${topicOf(GRADE_NAME[intent])} ${g.grade}등급(${g.score}점)이에요.` : NO_DATA
    }
    case 'type': {
      const [a, b] = d.types
      if (a.score < TYPE_MIN) {
        const score = (t: TypeName) => d.types.find((x) => x.name === t)?.score
        return `뚜렷하게 맞는 유형은 없어요. ${TYPE_NAMES.map((t) => `${t} ${score(t)}점`).join(' · ')}이에요.`
      }
      return `${a.name} ${a.score}점으로 가장 잘 맞고, 다음은 ${b.name} ${b.score}점이에요.`
    }
    case 'coast':
      return d.coastKm !== null && d.coastKm <= 1
        ? `해안선까지 ${d.coastKm.toFixed(1)}km인 바닷가 동네예요.`
        : '해안선에서 1km 넘게 떨어진 동네예요.'
    case 'life': {
      const life = allPins().get(d.id)?.life
      return life ? `${life} 성격이 강한 동네예요.` : NO_DATA
    }
    case 'similar':
      return d.similar.length > 0
        ? `지표가 비슷한 동네는 ${d.similar.map((s) => `${s.sigungu} ${s.name}`).join(', ')}이에요.`
        : NO_DATA
  }
}

/** 질문에 나온 다른 동 이름. 같은 시군구를 먼저, 긴 이름부터 본다 */
function mentioned(q: string, panelId: string, sigunguId: string) {
  const others = DONG_BASE.filter((b) => b.id !== panelId).sort(
    (a, b) =>
      Number(b.sigunguId === sigunguId) - Number(a.sigunguId === sigunguId) ||
      b.name.length - a.name.length,
  )
  return others.find((b) => q.includes(b.name)) ?? null
}

/** 핀 등급으로 만든 비교 문장 (조용함 · 저녁 활기). 값이 없는 축은 뺀다 */
function compareSentences(a: PinDong, b: PinDong): string[] {
  return (['quiet', 'evening'] as const)
    .filter((k) => a.grades[k] && b.grades[k])
    .map(
      (k) =>
        `${topicOf(GRADE_NAME[k])} ${a.name} ${a.grades[k]}등급, ${b.name} ${b.grades[k]}등급이에요.`,
    )
}

/**
 * POST /api/explore/chat mock. 대화 기억은 흉내 내지 않는다(매 질문 독립).
 * 같은 시군구 동 이름 → 비교 2문장, 다른 시군구 동 이름 → 같은 시군구 안내, 식당·카페 → 안내,
 * 낱말이 맞는 의도 → 규칙 문장, 그 밖 → 안내. 없는 동은 null (api 가 NotFoundError)
 */
export function chatMock(body: ChatRequest): ChatResponse | null {
  const dong = dongMock(body.panel)
  if (!dong) return null
  const q = body.question.trim()
  const base = {
    conversationId: body.conversationId ?? `mock-${dong.id}`,
    dongs: [{ id: dong.id, name: dong.name }],
  }
  const guide = (intent: string, text: string, examples: string[]): ChatResponse => ({
    ...base,
    intent,
    sentences: [],
    guide: text,
    examples,
  })

  const other = mentioned(q, dong.id, dong.sigungu.id)
  if (other && other.sigunguId !== dong.sigungu.id) {
    const near = DONG_BASE.filter((b) => b.sigunguId === dong.sigungu.id && b.id !== dong.id)
      .slice(0, 3)
      .map((b) => b.name)
    return guide(
      'compare_summary',
      `${SAME_SGG_ONLY} ${dong.sigungu.name}에서는 ${near.join(', ')} 등과 비교해 볼 수 있어요.`,
      [CANONICAL.quiet, CANONICAL.type, CANONICAL.similar],
    )
  }
  const pins = allPins()
  const a = pins.get(dong.id)
  const b = other && pins.get(other.id)
  if (a && b) {
    return {
      ...base,
      dongs: [...base.dongs, { id: b.id, name: b.name }],
      intent: 'compare_summary',
      sentences: compareSentences(a, b).map((text) => ({ text, source: 'rule' })),
      guide: null,
      examples: COMPARE_EXAMPLES,
    }
  }

  if (GUIDE_PLACE.pattern.test(q)) {
    return guide(GUIDE_PLACE.intent, GUIDE_PLACE.text, GUIDE_PLACE.examples)
  }
  const intent = KEYWORDS.find(([, re]) => re.test(q))?.[0]
  if (!intent) return guide(GUIDE_OTHER.intent, GUIDE_OTHER.text, GUIDE_OTHER.examples)
  return {
    ...base,
    intent,
    sentences: [{ text: ruleSentence(intent, dong), source: 'rule' }],
    guide: null,
    examples: INTENTS.filter((i) => i !== intent)
      .slice(0, 3)
      .map((i) => CANONICAL[i]),
  }
}
```

```ts
import { useQuery, useQueryClient } from '@tanstack/react-query'
import { postChat } from '@/features/explore/api'
import { EXPLORE_COPY } from '@/features/explore/copy'
import { exploreKeys } from '@/features/explore/queryKeys'
import type { ChatErrorKind, ChatThread, ChatTurn } from '@/features/explore/types'
import { chatInterest } from '@/features/interest'

/** api() 가 던지는 `API {status} …` 메시지에서 화면 분기를 고른다. 그 밖(네트워크·503·404)은 다시 시도 */
export function chatErrorKind(error: unknown): ChatErrorKind {
  const status = error instanceof Error ? /^API (\d{3})/.exec(error.message)?.[1] : undefined
  if (status === '429') return 'limit'
  if (status === '400') return 'invalid'
  return 'retry'
}

const emptyThread = (): ChatThread => ({
  conversationId: null,
  turns: [],
  examples: [...EXPLORE_COPY.chatExamples],
})

const withTurn = (t: ChatThread, turn: ChatTurn): ChatThread => ({
  ...t,
  turns: t.turns.map((x) => (x.id === turn.id ? turn : x)),
})

const isPending = (t: ChatThread) => t.turns.some((x) => x.status === 'pending')

let seq = 0

/**
 * ② 동 패널 채팅 기록 (웹 스펙 §4-2). 기록은 QueryClient 의 exploreKeys.chatThread(id) 에
 * 두어(gcTime Infinity, 서버 요청 없음) 동을 바꿨다 돌아와도 남고, 새로고침하면 사라진다.
 * 서버가 준 conversationId 를 그 동 기록에 두고 다음 질문에 실어 이어 묻는다.
 * 서버가 30분 지나 대화를 잊고 새 id 를 주면 화면 기록은 두고 id 만 바꾼다.
 * 응답이 오기 전에 동을 바꿔도 질문한 동의 기록에 답이 붙는다
 */
export function useChatThread(dongId: string) {
  const client = useQueryClient()
  const key = exploreKeys.chatThread(dongId)
  const { data: thread } = useQuery({
    queryKey: key,
    queryFn: emptyThread,
    initialData: emptyThread,
    staleTime: Infinity,
    gcTime: Infinity,
  })
  const pending = isPending(thread)

  /** 렌더 사이에 두 번 눌러도 한 번만 보내도록 캐시의 지금 값을 본다 */
  const current = () => client.getQueryData<ChatThread>(key) ?? thread
  const update = (fn: (t: ChatThread) => ChatThread) =>
    client.setQueryData<ChatThread>(key, (old) => fn(old ?? emptyThread()))

  async function run(id: string, question: string) {
    try {
      const res = await postChat({
        conversationId: current().conversationId,
        panel: dongId,
        question,
        criteria: { chips: [], interest: chatInterest() },
      })
      const done: ChatTurn = {
        id,
        question,
        status: 'done',
        sentences: res.sentences.map(({ text, source }) => ({ text, source })),
        guide: res.guide,
      }
      update((t) => ({
        ...withTurn(t, done),
        conversationId: res.conversationId,
        examples: res.examples.length > 0 ? res.examples : t.examples,
      }))
    } catch (error) {
      const failed: ChatTurn = { id, question, status: 'error', kind: chatErrorKind(error) }
      update((t) => withTurn(t, failed))
    }
  }

  /** 앞뒤 공백을 빼고 보낸다. 빈 질문이나 대기 중이면 무시 */
  function ask(raw: string) {
    const question = raw.trim()
    if (!question || isPending(current())) return
    const id = `chat-${++seq}`
    update((t) => ({ ...t, turns: [...t.turns, { id, question, status: 'pending' }] }))
    void run(id, question)
  }

  /** 실패한 차례를 그 자리에서 다시 묻는다 */
  function retry(id: string) {
    const now = current()
    const turn = now.turns.find((x) => x.id === id)
    if (!turn || turn.status !== 'error' || isPending(now)) return
    update((t) => withTurn(t, { id, question: turn.question, status: 'pending' }))
    void run(id, turn.question)
  }

  return { thread, pending, ask, retry }
}
```

```tsx
  )
```

```tsx
import { useEffect, useRef } from 'react'
import { EXPLORE_COPY } from '@/features/explore/copy'
import type { DongDetail } from '@/features/explore/types'
import { useChatThread } from '@/features/explore/useChatThread'
import Briefing from '@/features/explore/components/panel/Briefing'
import ChatBubble from '@/features/explore/components/panel/ChatBubble'
import ChatInput from '@/features/explore/components/panel/ChatInput'

type Props = { dong: DongDetail }

const CHIP =
  'border-border-default bg-bg-surface text-caption-xs text-text-primary hover:border-text-primary rounded-full border px-3 py-1.5 font-medium transition-colors disabled:cursor-not-allowed disabled:opacity-40'

/**
 * ② 동 패널 채팅 (웹 스펙 §3, Figma W-09 v7 466:9900). 브리핑 칸 안에 대화 · 예시 질문 · 입력을 붙인다.
 * 예시 질문은 늘 보이고 답이 오면 응답 examples 로 바뀐다. 대화 목록은 높이 288px 까지만 늘고
 * 그 안에서 스크롤된다(예시·입력은 아래에 고정). 새 차례가 붙거나 답이 오면 그 말풍선이 보이게
 * 내린다(동작 줄이기면 즉시). 돌아온 동의 기록에서는 내리지 않는다
 */
export default function ChatThread({ dong }: Props) {
  const { thread, pending, ask, retry } = useChatThread(dong.id)
  const last = thread.turns.at(-1)
  const lastRef = useRef<HTMLDivElement>(null)
  const seen = useRef(last)
  useEffect(() => {
    if (!last || last === seen.current) return
    seen.current = last
    const reduce = window.matchMedia?.('(prefers-reduced-motion: reduce)').matches
    lastRef.current?.scrollIntoView?.({ block: 'nearest', behavior: reduce ? 'auto' : 'smooth' })
  }, [last])

  return (
    <Briefing sentences={dong.briefing}>
      <div
        role="log"
        aria-live="polite"
        aria-label={EXPLORE_COPY.chatLog}
        tabIndex={0}
        className="flex max-h-72 flex-col gap-2 overflow-y-auto overscroll-contain empty:hidden"
      >
        {thread.turns.map((turn) => (
          <ChatBubble
            key={turn.id}
            ref={turn === last ? lastRef : undefined}
            turn={turn}
            onRetry={retry}
          />
        ))}
      </div>
      <div
        role="group"
        aria-label={EXPLORE_COPY.chatExamplesLabel}
        className="flex flex-wrap gap-1.5"
      >
        {thread.examples.map((q) => (
          <button key={q} type="button" disabled={pending} onClick={() => ask(q)} className={CHIP}>
            {q}
          </button>
        ))}
      </div>
      <ChatInput disabled={pending} onSend={ask} />
    </Briefing>
  )
}
```
