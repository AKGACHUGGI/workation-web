# W-08 마이페이지 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 로그인한 사용자의 즐겨찾기(지역·동네)·팁·사진·알림 설정·계정을 `/me`에서 관리하고, W-01·W-03·W-04의 하트 버튼이 쓰는 `useFavorites`를 서버 기반 실구현으로 교체한다.

**Architecture:** `features/me/`에 타입(백엔드 계약)·API·쿼리 훅·순수 함수·컴포넌트를 모은다. 즐겨찾기는 서버가 진실이며 `['me','favorites','ids']` 쿼리를 낙관적으로 갱신하고 실패 시 롤백한다. 비로그인 `/me`는 W-07 `useAuthContext().openLogin(undefined, { onDismiss })`로 모달을 열고, 로그인 없이 닫히면 `onDismiss`가 `/`로 보낸다 — W-07 `AuthProvider`는 고치지 않는다. `tab`은 URL 쿼리.

**Tech Stack:** React 19 · TanStack Query 5(`useQuery`, `useInfiniteQuery`, `useMutation` 낙관적 업데이트) · react-router-dom 7 · Vitest + Testing Library · Playwright

**Spec:** `docs/superpowers/specs/2026-11-07-W-08-마이페이지-design.md` (Figma 251:4679)

## Global Constraints

- LG U+ 기업연계 과제: 국내 타 기업 API 자제. 로그인은 카카오·Google(W-07)만.
- **즐겨찾기는 로그인 필수.** 비로그인 저장·localStorage·병합 없음. 비로그인 하트 클릭 → `useRequireLogin()` → 로그인 모달.
- 웹 데스크톱 1440만, 최소 폭 1128(`min-w-desktop`). 본문 맑은 고딕(공통 기반).
- 색·치수·타이포는 `@theme` 토큰만. 임의값 금지.
- 파일 1개 = 컴포넌트 1개 `export default function`. 컴포넌트 파일에서 상수 export 금지.
- `enum` 금지(`as const` + 유니온). 타입은 `import type`.
- 데이터 호출은 `lib/api.ts`의 `api<T>()`로만. mock은 `isMock` 분기.
- `pages/`는 다른 `pages/`를 import하지 않는다.
- 되돌리기 토스트 5초. 상한 50개 초과(409) → 토스트 "최대 50개까지 저장할 수 있어요".
- Conventional Commits. 커밋 전 `pnpm lint && pnpm format:check && pnpm build && pnpm test`.

import type { MeSummary } from '@/features/me/types'

/** '2025-03-14T…' → '2025.03' */
export function formatJoinedMonth(iso: string): string {
  return `${iso.slice(0, 4)}.${iso.slice(5, 7)}`
}

/** 프로필 헤더 한 줄: '강릉 2회 · 고성 1회 체류 · 팁 7개 · 사진 12장 · 2025.03 가입' */
export function buildActivitySummary(s: MeSummary): string {
  const stays =
    s.stays.length === 0
      ? '아직 체류 기록 없음'
      : `${s.stays.map((x) => `${x.regionName} ${x.count}회`).join(' · ')} 체류`
  return `${stays} · 팁 ${s.tipCount}개 · 사진 ${s.photoCount}장 · ${formatJoinedMonth(s.createdAt)} 가입`
}

import { useCallback } from 'react'
import { useMutation, useQuery, useQueryClient } from '@tanstack/react-query'
import { useToast } from '@/components/ui/toast/useToast'
import { useAuth } from '@/features/auth/useAuth'
import { useRequireLogin } from '@/features/auth/useRequireLogin'
import { FavoriteLimitError, deleteFavorite, getFavoriteIds, putFavorite } from '@/features/me/api'
import { ME_COPY } from '@/features/me/copy'
import { meKeys } from '@/features/me/queryKeys'
import type { FavoriteIds, FavoriteKind } from '@/features/me/types'

export type { FavoriteKind } from '@/features/me/types'

export type UseFavorites = {
  isFavorite: (kind: FavoriteKind, id: string) => boolean
  toggle: (kind: FavoriteKind, id: string) => void
  isPending: boolean
}

const EMPTY: FavoriteIds = { regions: [], dongs: [] }

type Vars = { kind: FavoriteKind; id: string; next: boolean }

/**
 * 즐겨찾기 상태·토글. 서버가 진실(['me','favorites','ids']).
 * 비로그인이면 toggle 은 로그인 모달만 연다. 로그인 상태면 낙관적 갱신 → PUT/DELETE → 실패 시 롤백.
 */
export function useFavorites(): UseFavorites {
  const { isLoggedIn } = useAuth()
  const requireLogin = useRequireLogin()
  const queryClient = useQueryClient()
  const { show } = useToast()

  const { data } = useQuery({
    queryKey: meKeys.favoriteIds,
    queryFn: getFavoriteIds,
    enabled: isLoggedIn,
  })
  const ids = data ?? EMPTY

  const mutation = useMutation({
    mutationFn: ({ kind, id, next }: Vars) =>
      next ? putFavorite(kind, id) : deleteFavorite(kind, id),
    onMutate: async ({ kind, id, next }) => {
      await queryClient.cancelQueries({ queryKey: meKeys.favoriteIds })
      const prev = queryClient.getQueryData<FavoriteIds>(meKeys.favoriteIds) ?? EMPTY
      const list = next ? [...prev[kind], id] : prev[kind].filter((x) => x !== id)
      queryClient.setQueryData<FavoriteIds>(meKeys.favoriteIds, { ...prev, [kind]: list })
      return { prev }
    },
    onError: (err, _vars, ctx) => {
      if (ctx) queryClient.setQueryData(meKeys.favoriteIds, ctx.prev)
      if (err instanceof FavoriteLimitError) show(ME_COPY.limit)
    },
    onSettled: () => {
      queryClient.invalidateQueries({ queryKey: meKeys.favoriteIds })
      queryClient.invalidateQueries({ queryKey: ['me', 'favorites'] })
      queryClient.invalidateQueries({ queryKey: meKeys.summary })
    },
  })

  const isFavorite = useCallback((kind: FavoriteKind, id: string) => ids[kind].includes(id), [ids])

  const toggle = useCallback(
    (kind: FavoriteKind, id: string) => {
      requireLogin(() => mutation.mutate({ kind, id, next: !ids[kind].includes(id) }))
    },
    [requireLogin, mutation, ids],
  )

  return { isFavorite, toggle, isPending: mutation.isPending }
}
```

import { useState } from 'react'

type Props = {
  initial: string
  onSave: (nickname: string) => Promise<void>
  onCancel: () => void
}

const MIN = 2
const MAX = 12

/** 닉네임 인라인 편집 (2~12자) */
export default function NicknameEditor({ initial, onSave, onCancel }: Props) {
  const [value, setValue] = useState(initial)
  const [saving, setSaving] = useState(false)
  const trimmed = value.trim()
  const valid = trimmed.length >= MIN && trimmed.length <= MAX

  async function save() {
    if (!valid) return
    setSaving(true)
    try {
      await onSave(trimmed)
    } finally {
      setSaving(false)
    }
  }

  return (
    <form
      className="flex items-center gap-2"
      onSubmit={(e) => {
        e.preventDefault()
        void save()
      }}
    >
      <input
        aria-label="닉네임"
        value={value}
        maxLength={MAX}
        onChange={(e) => setValue(e.target.value)}
        className="border-border-default text-body-base text-text-primary h-10 rounded-lg border px-3"
      />
      <button
        type="submit"
        disabled={!valid || saving}
        className="bg-brand-primary text-body-sm text-text-inverse hover:bg-brand-hover disabled:opacity-dim rounded-lg px-4 py-2"
      >
        저장
      </button>
      <button type="button" onClick={onCancel} className="text-body-sm text-text-secondary px-2">
        취소
      </button>
      {!valid && <span className="text-caption-xs text-signal-avoid">2~12자로 입력해 주세요</span>}
    </form>
  )
}

/** 카드 Meta2: 메모 → 방문 예정 월 → 시기 여유 */
export function favoriteMeta2(r: FavoriteRegion, month: MonthParam): string {
  if (r.note) return r.note
  if (r.visitMonth) return `${monthLabel(r.visitMonth)} 방문 예정`
  return `${monthLabel(month)} 시기 여유 ${metricGrade(r.scores.crowd)} · 혼잡 ${r.scores.crowd}`
}

/** preset 쿼리 → 가중치. custom 인데 metrics 가 없으면 조용한 곳 */
export function weightsFor(preset: PresetKey, metrics: MetricKey[] | null): Weights {
  if (preset === 'custom') return customWeights(metrics ?? ['quiet', 'crowd', 'overnight'])
  return PRESET_WEIGHTS[preset]
}

import type { MyTab } from '@/features/me/components/tabs'

import { useMyFavorites } from '@/features/me/useMyFavorites'
import { parseMetrics } from '@/features/metric/params'
import { useMonthParam, usePresetParam, useQueryState } from '@/hooks/useQueryState'

/** W-08 마이페이지 (로그인 필수) */
export default function MyPage() {
  const { me, isPending: authPending, isLoggedIn } = useAuth()
  const { openLogin } = useAuthContext()
  const navigate = useNavigate()
  const [tab, setTab] = useQueryState<MyTab>('tab', parseMyTab, DEFAULT_MY_TAB)
  const [month] = useMonthParam()
  const [preset] = usePresetParam()
  const [metrics] = useQueryState('metrics', parseMetrics, null)

  // 비로그인: 모달을 열고, 로그인 없이 닫히면 홈으로
  useEffect(() => {
    if (authPending || isLoggedIn) return
    openLogin(undefined, { onDismiss: () => navigate('/', { replace: true }) })
  }, [authPending, isLoggedIn, openLogin, navigate])

  const summary = useMeSummary()
  const favorites = useMyFavorites(month)

  if (authPending || !isLoggedIn || !me) {
    return (
      <PageContainer className="py-10">
        <h1 className="sr-only">마이페이지</h1>
      </PageContainer>
    )
  }
  if (summary.isPending) {
    return (
      <PageContainer className="py-10">
        <MySkeleton />
      </PageContainer>
    )
  }
  if (summary.isError) {
    return (
      <PageContainer className="py-10">
        <p className="text-body-base text-text-secondary">
          마이페이지를 불러오지 못했습니다 ·{' '}
          <button type="button" onClick={() => summary.refetch()} className="underline">
            다시 시도
          </button>
        </p>
      </PageContainer>
    )
  }

  const fav = favorites.data
  const favEmpty = fav && fav.regions.length === 0 && fav.dongs.length === 0

  return (
    <PageContainer className="flex flex-col gap-6 py-10">
      <ProfileHeader me={me} summary={summary.data} />
      <MyTabs
        value={tab}
        onChange={setTab}
        tipCount={summary.data.tipCount}
        photoCount={summary.data.photoCount}
      />
      <div className="grid grid-cols-3 gap-6">
        <div className="col-span-2 flex flex-col gap-10">
          {tab === 'favorites' &&
            (favorites.isPending ? (
              <div className="bg-bg-subtle h-64 animate-pulse rounded-xl" />
            ) : favorites.isError || !fav ? (
              <p className="text-body-sm text-text-secondary">
                불러오지 못했습니다 ·{' '}
                <button type="button" onClick={() => favorites.refetch()} className="underline">
                  다시 시도
                </button>
              </p>
            ) : favEmpty ? (
              <FavoritesEmpty />
            ) : (
              <>
                {fav.regions.length > 0 && (
                  <FavoriteRegions
                    regions={fav.regions}
                    month={month}
                    preset={preset}
                    metrics={metrics}
                  />
                )}
                {fav.dongs.length > 0 && <FavoriteDongs dongs={fav.dongs} />}
              </>
            ))}
          {tab === 'tips' && <MyTips />}
          {tab === 'photos' && <MyPhotos />}
        </div>
        <aside className="flex flex-col gap-6">
          <NotificationSettings />
          <AccountSection me={me} summary={summary.data} />
        </aside>
      </div>
    </PageContainer>
  )
}
```

`useQueryState('metrics', parseMetrics, null)`의 `T`는 `MetricKey[] | null`이다. `parse`가 `null`을 돌려주면 "무효"로 간주되어 기본값(`null`)으로 replace 되는데, 결과가 같으므로 URL이 `metrics=`로 덮이지 않도록 **키가 있고 무효일 때만** 보정된다(공통 훅 동작). 문제가 되면 `[metrics]`를 `useSearchParams().get('metrics')`에 `parseMetrics`를 직접 적용하는 방식으로 바꾼다.
