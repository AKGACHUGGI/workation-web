# W-06 지자체 진단 리포트 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** `/reports/:sggId`에서 시군구 하나의 성수기 집중도·숙박 성향·대상 지역 대비 위치·월별 방문 지수·진단 요약을 연간(2025) 기준으로 보여주고, 백엔드가 만든 PDF를 내려받는다.

**Architecture:** 연간 집계·순위·진단 문장은 모두 백엔드 응답(`ReportResponse`)이며 프론트는 표시와 PDF 다운로드만 한다. `features/report/`에 타입·API·훅·문구·컴포넌트를 두고 `pages/ReportPage.tsx`가 조립한다. URL 상태는 `preset`(`quiet`|`lively`, `custom`은 `quiet`로 보정)과 `year`(2025 고정). 차트는 W-03 `MonthlyCrowdChart`와 같은 Recharts `BarChart` 설정을 공유한다.

**Tech Stack:** React 19 · TypeScript 6 · TanStack Query 5 · react-router-dom 7 · Recharts 3 · lucide-react · Tailwind v4 토큰 · Vitest + Testing Library · Playwright

파일 끝에 아래를 추가한다(기존 `API_BASE_URL`·`api<T>`는 그대로). 같은 쿠키·CSRF 헤더·에러 규칙을 따르되, 본문을 JSON 으로 읽지 않고 `Content-Type` 도 보내지 않는다(GET 이라 본문 없음):

```ts
/**
 * 파일(blob) 응답용. PDF 다운로드처럼 JSON 이 아닌 응답에만 쓴다.
 * 컴포넌트에서 fetch 를 직접 쓰지 않기 위한 유일한 예외 경로. 쿠키·CSRF·에러 규칙은 api<T>() 와 같다.
 */
export async function apiBlob(path: string, init?: RequestInit): Promise<Blob> {
  const res = await fetch(`${API_BASE_URL}${path}`, {
    ...init,
    credentials: 'include',
    headers: {
      'X-Requested-With': 'workation-web',
      ...init?.headers,
    },
  })
  if (res.status === 401) throw new UnauthorizedError(path)
  if (res.status === 404) throw new NotFoundError(path)
  if (!res.ok) throw new Error(`API ${res.status} ${res.statusText}: ${path}`)
  return res.blob()
}
```

```ts
import type { ReportPreset, ReportResponse } from '@/features/report/types'
```

```ts
const IDS = Object.keys(SIGUNGU_BASE)
const TOTAL = IDS.length

/** 값 내림차순 순위 (1부터). 동점은 같은 순위 */
function rankOf(id: string, pick: (sid: string) => number): number {
  const mine = pick(id)
  return IDS.filter((sid) => pick(sid) > mine).length + 1
}

/** 순위 → 0~100 백분위 (1위 92, 2위 83 … 12위 0) */
function percentile(rank: number): number {
  return Math.round((100 * (TOTAL - rank)) / TOTAL)
}

function topMonths(months: number[]): number[] {
  return months
    .map((crowd, i) => ({ month: i + 1, crowd }))
    .sort((a, b) => b.crowd - a.crowd || a.month - b.month)
    .slice(0, 3)
    .map((m) => m.month)
    .sort((a, b) => a - b)
}

function narrative(seasonType: SeasonType, months: number[]): string {
  const top = formatMonths(topMonths(months))
  const low = months
    .map((c, i) => ({ c, m: i + 1 }))
    .filter((x) => x.c <= 50)
    .map((x) => x.m)
  const lowCopy = low.length
    ? `${formatMonths(low)}은 연중 최대의 절반 이하로, 비수기 체류 지원에 적합합니다.`
    : '연중 방문이 고르게 분포해 특정 비수기가 뚜렷하지 않습니다.'
  const head: Record<SeasonType, string> = {
    summer_beach: `${top}에 방문이 집중되는 여름 해변형입니다.`,
    spring_autumn: `${top}에 방문이 몰리는 봄·가을형입니다.`,
    second_half_rise: `7·8월 여름 성수기 뒤 10월에 다시 오르는 하반기 상승형입니다.`,
    flat: `${top}이 높지만 연중 편차가 작은 연중 평탄형입니다.`,
  }
  return `${head[seasonType]} ${lowCopy}`
}

function diagnosis(
  name: string,
  ranks: ReportResponse['metricRanks'],
  coef: number,
  peak: number,
  avg: number,
) {
  const strong = METRIC_KEYS.reduce((b, k) => (ranks[k].rank < ranks[b].rank ? k : b))
  const weak = METRIC_KEYS.reduce((w, k) => (ranks[k].rank > ranks[w].rank ? k : w))
  return {
    strength: `${METRIC_LABEL[strong]} ${ranks[strong].rank}위로 12곳 중 상위권이며, 숙박 계수 ${coef.toFixed(2)}로 ${overnightCopy(coef)} 유형입니다.`,
    weakness: `${METRIC_LABEL[weak]}이 ${ranks[weak].rank}위로 가장 약하고, 성수기 집중도 ${peak}%는 12곳 평균 ${avg}%${peak >= avg ? '보다 높아 특정 달에 방문이 쏠립니다' : '보다 낮아 성수기 효과가 작습니다'}.`,
    suggestion: `${name}은 ${METRIC_LABEL[strong]} 강점을 내세운 비수기 체류형 워케이션 프로그램이 적합하며, ${METRIC_LABEL[weak]} 보완 시 연간 순위 상승 여지가 있습니다.`,
  }
}
```

```ts
import { useCallback, useRef, useState } from 'react'
import { useToast } from '@/components/ui/toast/useToast'
import { fetchReportPdf } from '@/features/report/api'
import { PDF_FAIL_MESSAGE, reportFileName } from '@/features/report/copy'
import type { ReportPreset } from '@/features/report/types'

type Params = { year: number; preset: ReportPreset; name: string }

/**
 * PDF 내보내기. blob → object URL → <a download> 클릭 → URL 해제.
 * 진행 중에는 재호출을 무시한다 (버튼은 isPending 으로 비활성).
 */
export function useReportPdf(id: string, { year, preset, name }: Params) {
  const [isPending, setPending] = useState(false)
  const pendingRef = useRef(false)
  const { show } = useToast()

  const download = useCallback(async () => {
    if (pendingRef.current) return
    pendingRef.current = true
    setPending(true)
    try {
      const blob = await fetchReportPdf(id, { year, preset })
      const url = URL.createObjectURL(blob)
      const a = document.createElement('a')
      a.href = url
      a.download = reportFileName(name, year)
      document.body.appendChild(a)
      a.click()
      a.remove()
      URL.revokeObjectURL(url)
    } catch {
      show(PDF_FAIL_MESSAGE)
    } finally {
      pendingRef.current = false
      setPending(false)
    }
  }, [id, year, preset, name, show])

  return { download, isPending }
}
```

```tsx
            const barTone = isCurrent ? 'bg-brand-primary' : 'bg-chart-bar-muted'
            return (
              <tr
                key={item.id}
                aria-current={isCurrent || undefined}
                className={`rounded-lg ${rowTone}`}
              >
                <td className="text-body-sm w-8 py-2 pl-3 font-medium">{i + 1}</td>
                <td className="text-body-sm w-24 py-2">
                  <Link to={carryQuery(`/reports/${item.id}`, current)} className="hover:underline">
                    {item.name}
                  </Link>
                </td>
                <td className="py-2 pr-3">
                  <div className="bg-bg-surface-strong h-2 w-full overflow-hidden rounded-full">
                    <div
                      data-testid={`bar-${item.id}`}
                      className={`h-full rounded-full ${barTone}`}
                      style={{ width: `${item.score}%` }}
                    />
                  </div>
                </td>
                <td className="text-body-sm w-10 py-2 pr-3 text-right font-medium">{item.score}</td>
              </tr>
            )
          })}
        </tbody>
      </table>
    </section>
  )
}
```

```tsx
import PageContainer from '@/components/layout/PageContainer'
import AnnualCrowdChart from '@/features/report/components/AnnualCrowdChart'
import DiagnosisSection from '@/features/report/components/DiagnosisSection'
import RankingTable from '@/features/report/components/RankingTable'
import ReportHeader from '@/features/report/components/ReportHeader'
import ReportSkeleton from '@/features/report/components/ReportSkeleton'
import SummaryCards from '@/features/report/components/SummaryCards'
import { REPORT_SOURCE_NOTE } from '@/features/report/copy'
import {
  DEFAULT_REPORT_PRESET,
  parseReportPreset,
  parseYear,
  REPORT_YEAR,
} from '@/features/report/params'
import { useReportPdf } from '@/features/report/useReportPdf'
import { useReportQuery } from '@/features/report/useReportQuery'
import { useQueryState } from '@/hooks/useQueryState'
import { NotFoundError } from '@/lib/errors'

/** W-06 지자체 진단 리포트. /reports/:sggId?preset=quiet|lively&year=2025 */
export default function ReportPage() {
  const { sggId = '' } = useParams()
  const [preset] = useQueryState('preset', parseReportPreset, DEFAULT_REPORT_PRESET)
  const [year] = useQueryState('year', parseYear, REPORT_YEAR)
  const query = useReportQuery(sggId, { year, preset })
  const pdf = useReportPdf(sggId, { year, preset, name: query.data?.name ?? '' })

  // 없는 sggId → 라우트의 errorElement(NotFoundPage) 가 레이아웃 안에서 받는다
  if (query.isError && query.error instanceof NotFoundError) throw query.error

  return (
    <PageContainer className="flex flex-col gap-6 py-10">
      <ReportHeader
        currentId={sggId}
        onExportPdf={pdf.download}
        isExporting={pdf.isPending || !query.data}
      />

      {query.isPending && <ReportSkeleton />}

      {query.isError && (
        <div
          role="alert"
          className="border-border-default flex flex-col items-center gap-3 rounded-xl border p-10"
        >
          <p className="text-body-base text-text-primary">리포트를 불러오지 못했습니다</p>
          <button
            type="button"
            onClick={() => query.refetch()}
            className="border-border-default text-body-sm text-text-primary hover:border-border-strong rounded-lg border px-4 py-2"
          >
            다시 시도
          </button>
        </div>
      )}

      {query.data && (
        <div className={`flex flex-col gap-6 ${query.isPlaceholderData ? 'opacity-dim' : ''}`}>
          <SummaryCards report={query.data} />
          <div className="grid grid-cols-5 gap-6">
            <div className="col-span-3">
              <AnnualCrowdChart
                months={query.data.months}
                topMonths={query.data.peakConcentration.topMonths}
                narrative={query.data.seasonNarrative}
              />
            </div>
            <div className="col-span-2">
              <RankingTable
                items={query.data.ranking.items}
                currentId={query.data.id}
                preset={preset}
              />
            </div>
          </div>
          <DiagnosisSection diagnosis={query.data.diagnosis} />
          <p className="text-caption-xs text-text-tertiary">{REPORT_SOURCE_NOTE}</p>
        </div>
      )}
    </PageContainer>
  )
}
```
